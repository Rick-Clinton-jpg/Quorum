# Runtime dependencies for service/main.py, deployed as the Cloud Run
# image. Combines what worker_agent/, gate/'s Quorum-owned code, and
# intent_graph actually need at runtime - not a copy of any single
# component's own requirements.txt.
#
# Deliberately excludes sentence-transformers (listed in
# verifiers/intent_graph/requirements.txt): it pulls in torch, which is
# a heavy, slow addition to a Cloud Run image, and intent_layer's own
# extractor.py already falls back automatically (and loudly - a logged
# warning) to an offline scikit-learn HashingVectorizer if the real
# model can't load - see docs/INTEGRATION_MAP.md Section 5's disclosure
# of this exact tradeoff. Add sentence-transformers back here if real
# semantic embeddings turn out to matter enough to justify the image-size
# and cold-start cost once this is actually deployed.

fastapi>=0.110.0
uvicorn[standard]>=0.28.0

google-adk>=2.7.1
google-genai>=2.18.1
pydantic>=2.0
python-dotenv>=1.0

google-cloud-firestore>=2.14.0

numpy
scikit-learn

# gate/github_action.py's PR-creation call (git CLI, already on the base
# image, handles clone/branch/apply/commit/push).
requests

# gate/otel_tracing.py - exports each gate stage as a real OTel span to
# Cloud Trace (falls back to console export if Cloud Trace isn't
# reachable - see that module's docstring).
opentelemetry-api>=1.24.0
opentelemetry-sdk>=1.24.0
opentelemetry-exporter-gcp-trace>=1.6.0

# worker_agent/self_check.py runs `sys.executable -m pytest` against a
# temp copy of verifiers/sentry - that's THIS process's own interpreter,
# so pytest has to be installed here too, not just in a dev environment.
# Confirmed live: without it, self_check fails with "No module named
# pytest" on every proposal in the deployed service. verifiers/sentry's
# own pyproject.toml sets pythonpath=["src"] as a pytest ini option
# (pytest>=7.0 native), so sentry.engine is importable without a
# separate install of the sentry package itself.
pytest>=7.0

# FastAPI's TestClient (service/tests/test_main.py, Agent Identity's
# HTTP-layer tests) needs httpx even though the app itself never calls
# out over HTTP - not a runtime dependency of main.py, just of testing it
# with the same interpreter self_check.py already requires pytest for.
httpx>=0.27.0
